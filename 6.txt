    <script>
        var arr = [];

        function showArray() {
            document.getElementById("display").textContent =
                "[" + arr.join(", ") + "]";
        }

        // Create Array
        function createArray() {
            var str = document.getElementById("elements").value;

            if (str.trim() == "") {
                alert("Enter array elements!");
                return;
            }

            var temp = str.split(",");

            for (var i = 0; i < temp.length; i++) {
                temp[i] = temp[i].trim();

                if (temp[i] == "") {
                    alert("Invalid array elements!");
                    return;
                }
            }

            arr = temp;
            showArray();
        }

        // Remove using splice()
        function removeMethod() {
            var x = document.getElementById("remove").value.trim();
            var index = arr.indexOf(x);

            if (index == -1) {
                document.getElementById("removeResult").innerHTML =
                    "Element not found!";
            } else {
                arr.splice(index, 1);
                document.getElementById("removeResult").innerHTML =
                    "Element removed!";
            }

            showArray();
        }

        // Remove without method
        function removeLoop() {
            var x = document.getElementById("remove").value.trim();
            var found = false;

            for (var i = 0; i < arr.length; i++) {
                if (arr[i] == x) {
                    for (var j = i; j < arr.length - 1; j++) {
                        arr[j] = arr[j + 1];
                    }

                    arr.length--;
                    found = true;
                    break;
                }
            }

            if (found) {
                document.getElementById("removeResult").innerHTML =
                    "Element removed!";
            } else {
                document.getElementById("removeResult").innerHTML =
                    "Element not found!";
            }

            showArray();
        }

        // Search using includes()
        function searchMethod() {
            var x = document.getElementById("search").value.trim();

            if (arr.includes(x)) {
                document.getElementById("searchResult").innerHTML =
                    "Element found!";
            } else {
                document.getElementById("searchResult").innerHTML =
                    "Element not found!";
            }
        }

        // Search without method
        function searchLoop() {
            var x = document.getElementById("search").value.trim();
            var found = false;

            for (var i = 0; i < arr.length; i++) {
                if (arr[i] == x) {
                    found = true;
                    break;
                }
            }

            if (found) {
                document.getElementById("searchResult").innerHTML =
                    "Element found!";
            } else {
                document.getElementById("searchResult").innerHTML =
                    "Element not found!";
            }
        }

        // Empty using length
        function emptyMethod() {
            arr.length = 0;
            showArray();

            document.getElementById("emptyResult").innerHTML =
                "Array emptied!";
        }

        // Empty without method
        function emptyArray() {
            arr = [];
            showArray();

            document.getElementById("emptyResult").innerHTML =
                "Array emptied!";
        }
    </script>
