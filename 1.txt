    <label>Length:</label>

    <label>Width:</label>
    <input type="number" id="width" placeholder="Enter width">

    <button onclick="rectangle()">Calculate Rectangle Area</button>

    <h2>3. Area of Circle</h2>

    <label>Radius:</label>
    <input type="number" id="radius" placeholder="Enter radius">
    <br><br>

    <button onclick="circle()">Calculate Circle Area</button>
    <p id="circleResult"></p>

    <hr>

    <script>
        function triangle() {
            var b = document.getElementById("base").value;
            var h = document.getElementById("height").value;

            if (b == "" || h == "" || b <= 0 || h <= 0) {
                document.getElementById("triangleResult").innerHTML =
                    "Please enter valid positive values.";
            } else {
                var area = 0.5 * b * h;
                document.getElementById("triangleResult").innerHTML =
                    "Area of Triangle = " + area;
            }
        }

        function rectangle() {
            var l = document.getElementById("length").value;
            var w = document.getElementById("width").value;

            if (l == "" || w == "" || l <= 0 || w <= 0) {
                document.getElementById("rectangleResult").innerHTML =
                    "Please enter valid positive values.";
            } else {
                var area = l * w;
                document.getElementById("rectangleResult").innerHTML =
                    "Area of Rectangle = " + area;
            }
        }

        function circle() {
            var r = document.getElementById("radius").value;

            if (r == "" || r <= 0) {
                document.getElementById("circleResult").innerHTML =
                    "Please enter a valid positive radius.";
            } else {
                var area = Math.PI * r * r;
                document.getElementById("circleResult").innerHTML =
                    "Area of Circle = " + area.toFixed(2);
            }
        }
    </script>
