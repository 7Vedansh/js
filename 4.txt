    <script>
        function compareStrings() {
            var a = document.getElementById("str1").value;
            var b = document.getElementById("str2").value;
            var result = "";

            if (a.trim() == "" || b.trim() == "") {
                document.getElementById("result").innerHTML =
                    "Please enter both strings.";
                return;
            }

            // 1. Length Comparison
            result += "<p>1. Length Equal: " +
                      (a.length == b.length) + "</p>";

            // 2. Loose Equality
            result += "<p>2. Loose Equality (==): " +
                      (a == b) + "</p>";

            // 3. Strict Equality
            result += "<p>3. Strict Equality (===): " +
                      (a === b) + "</p>";

            // 4. Locale Comparison
            var x = a.localeCompare(b);

            if (x == 0) {
                result += "<p>4. Locale Compare: Equal</p>";
            } else if (x < 0) {
                result += "<p>4. Locale Compare: First comes before second</p>";
            } else {
                result += "<p>4. Locale Compare: First comes after second</p>";
            }

            // 5. Case-Insensitive Comparison
            if (a.toLowerCase() == b.toLowerCase()) {
                result += "<p>5. Case-Insensitive: Equal</p>";
            } else {
                result += "<p>5. Case-Insensitive: Not Equal</p>";
            }

            document.getElementById("result").innerHTML = result;
        }
    </script>
