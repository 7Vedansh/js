    <h2>1. Mouse Over Event</h2>

    <div id="mouseBox"
         onmouseover="mouseOverEvent()"
         onmouseout="mouseOutEvent()">

        <h3>Move your mouse here</h3>

    </div>

    <hr>

    <h2>2. Focus Event</h2>

    <label>Enter Your Name:</label>

    <input type="text" id="name"
           onfocus="focusEvent()"
           onblur="blurEvent()">

    <h3 id="output"></h3>

    <script>
        function mouseOverEvent() {
            document.getElementById("mouseBox").style.backgroundColor = "yellow";
            document.getElementById("output").innerHTML =
                "Mouse Over Event Occurred";
        }

        function mouseOutEvent() {
            document.getElementById("mouseBox").style.backgroundColor = "white";
            document.getElementById("output").innerHTML =
                "Mouse Out Event Occurred";
        }

        function focusEvent() {
            document.getElementById("name").style.backgroundColor = "lightblue";
            document.getElementById("output").innerHTML =
                "Focus Event Occurred";
        }

        function blurEvent() {
            document.getElementById("name").style.backgroundColor = "white";
            document.getElementById("output").innerHTML =
                "Blur Event Occurred";
        }
    </script>
