    <script>
        function forTable() {
            var n = document.getElementById("num1").value;
            var limit = document.getElementById("limit1").value;
            var result = "";

            if (n == "" || limit == "" || limit <= 0 ||
                !Number.isInteger(Number(limit))) {
                result = "Please enter valid inputs.";
            } else {
                for (var i = 1; i <= limit; i++) {
                    result += n + " x " + i + " = " + (n * i) + "<br>";
                }
            }

            document.getElementById("result1").innerHTML = result;
        }

        function whileTable() {
            var n = document.getElementById("num2").value;
            var limit = document.getElementById("limit2").value;
            var result = "";
            var i = 1;

            if (n == "" || limit == "" || limit <= 0 ||
                !Number.isInteger(Number(limit))) {
                result = "Please enter valid inputs.";
            } else {
                while (i <= limit) {
                    result += n + " x " + i + " = " + (n * i) + "<br>";
                    i++;
                }
            }

            document.getElementById("result2").innerHTML = result;
        }

        function doWhileTable() {
            var n = document.getElementById("num3").value;
            var limit = document.getElementById("limit3").value;
            var result = "";
            var i = 1;

            if (n == "" || limit == "" || limit <= 0 ||
                !Number.isInteger(Number(limit))) {
                result = "Please enter valid inputs.";
            } else {
                do {
                    result += n + " x " + i + " = " + (n * i) + "<br>";
                    i++;
                } while (i <= limit);
            }

            document.getElementById("result3").innerHTML = result;
        }
